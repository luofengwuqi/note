# 跨域资源共享（CORS）策略拦截导致请求失败
浏览器的 “同源策略” 限制了不同源（协议 / 域名 / 端口任意一个不同）的请求，你的前端（本地 HTML 文件，协议是file://）与后端（http://localhost:8080）属于不同源，因此浏览器拦截了请求（核心是后端响应未携带Access-Control-Allow-Origin头）。  
## 解决方法:添加以下的配置类
```java
package com.example.demo.config;

import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.CorsRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

@Configuration
public class CorsConfig implements WebMvcConfigurer {
    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/**") // 允许所有后端接口跨域
                .allowedOrigins("*") // 允许所有前端源（开发环境用，生产环境需指定具体域名）
                .allowedMethods("GET", "POST", "PUT", "DELETE") // 允许的请求方法
                .allowedHeaders("*") // 允许的请求头
                .maxAge(3600); // 预检请求的缓存时间（秒）
    }
}
```

### 实例化axios:utils/request.js
创建 axios 实例，将来对创建出来的实例，进行自定义配置  
好处：不会污染原始的 axios 实例
```javascript
import router from "@/router";
import axios from "axios";
import { useUserStore } from "@/stores";
import { ElMessage } from "element-plus";
//1.创建axios实例
const instance = axios.create({
  //1.1基地址
  baseURL: 'http://localhost:8080/',
  //1.2超时时间设定(单位:ms)
  timeout: 5000,
});
//2自定义配置 - 请求/相应拦截器
    // 2.1添加请求拦截器
    instance.interceptors.request.use(function (config) {
    // 在发送请求之前做些什么()

    //例:携带token
    const userStore = useUserStore()
    if(userStore.token){
        config.headers.Authorization = userStore.token
    }

    return config;
  }, function (error) {
    // 对请求错误做些什么
    return Promise.reject(error);
  });

    // 2.2添加响应拦截器
    instance.interceptors.response.use(function (response) {
    // 对响应数据做点什么
    if(response.data.code === 0){
        return response.data;
    }
    //处理失败
    ElMessage.error(response.data.message || '服务异常...')//用Element的提示框报错
    return Promise.reject(response.data)
  }, function (error) {
    //错误的特殊情况
    if(error.response?.status === 401){
        ElMessage.warning('登录状态已过期，请重新登录'); 
        router.push('/login')
    }

    //错误的默认情况
    ElMessage.error(error.response?.data?.message || '服务异常,请稍后再试')//用Element的提示框报错
    return Promise.reject(error);
  });

//3.导出
export default instance
```
